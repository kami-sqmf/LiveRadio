cmake_minimum_required(VERSION 3.20)
project(AirplayRadioNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED libplist-2.0 openssl)
if(NOT CODECS_ROOT)
  message(FATAL_ERROR "Set CODECS_ROOT to the minimal static FFmpeg installation (scripts/build-codecs.sh)")
endif()
set(UPSTREAM ${CMAKE_CURRENT_SOURCE_DIR}/vendor/uxplay-lib)
file(GLOB RAOP_SOURCES ${UPSTREAM}/*.c ${UPSTREAM}/playfair/*.c ${UPSTREAM}/llhttp/*.c)
list(FILTER RAOP_SOURCES EXCLUDE REGEX "/dnssd.c$")
add_library(raop_core STATIC vendor/dnssd_embedded.c ${RAOP_SOURCES})
target_include_directories(raop_core PUBLIC ${UPSTREAM} ${UPSTREAM}/playfair ${UPSTREAM}/llhttp ${DEPS_INCLUDE_DIRS} ${CODECS_ROOT}/include)
target_compile_definitions(raop_core PUBLIC WIN32 PLIST_210 PLIST_230 LIBPLIST_STATIC OPENSSL_API_COMPAT=0x10101000L)
find_library(STATIC_CRYPTO NAMES libcrypto.a REQUIRED)
find_library(STATIC_PLIST NAMES libplist-2.0.a REQUIRED)
find_library(STATIC_PTHREAD NAMES libwinpthread.a REQUIRED)
target_link_libraries(raop_core PUBLIC ${CODECS_ROOT}/lib/libavcodec.a ${CODECS_ROOT}/lib/libswresample.a ${CODECS_ROOT}/lib/libavutil.a ${STATIC_CRYPTO} ${STATIC_PLIST} ${STATIC_PTHREAD} ws2_32 iphlpapi crypt32 gdi32 bcrypt ole32 uuid)
add_library(AirplayRadioNative SHARED receiver.cpp)
target_link_libraries(AirplayRadioNative PRIVATE raop_core)
set_target_properties(AirplayRadioNative PROPERTIES PREFIX "")
target_link_options(AirplayRadioNative PRIVATE -static-libgcc -static-libstdc++)
add_executable(receiver_checks ../tests/receiver_checks.cpp)
target_link_libraries(receiver_checks PRIVATE AirplayRadioNative ws2_32)
target_link_options(receiver_checks PRIVATE -static)
add_executable(decoder_checks ../tests/decoder_checks.cpp)
target_link_libraries(decoder_checks PRIVATE raop_core)
target_link_options(decoder_checks PRIVATE -static-libgcc -static-libstdc++)
target_compile_definitions(decoder_checks PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../tests/fixtures")
enable_testing()
add_test(NAME receiver_checks COMMAND receiver_checks)
add_test(NAME decoder_checks COMMAND decoder_checks)
